🗄️ Supabase Graph Schema — Run in SQL Editor

-- NODES: entities extracted from cases
CREATE TABLE kg_nodes (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  ref TEXT NOT NULL,
  entity_type TEXT, -- person, company, property, account, grant
  entity_name TEXT,
  attributes JSONB,
  domain TEXT, -- investigation, zero_rent, financial, telecom
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- EDGES: relationships between entities
CREATE TABLE kg_edges (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  source_id UUID REFERENCES kg_nodes(id),
  target_id UUID REFERENCES kg_nodes(id),
  relation_type TEXT, -- owns, paid, connected_to, applied_for, lives_at
  weight FLOAT DEFAULT 1.0,
  evidence TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Enable RLS
ALTER TABLE kg_nodes ENABLE ROW LEVEL SECURITY;
ALTER TABLE kg_edges ENABLE ROW LEVEL SECURITY;

-- 1. INVESTIGATION CASES (основна таблица)

  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  branch TEXT CHECK (branch IN ('UK','USA','Dubai')),
  source_channel TEXT,
  target_name TEXT,
  asset_identifier TEXT UNIQUE,
  market_value NUMERIC DEFAULT 0,
  outstanding_debt NUMERIC DEFAULT 0,
  status TEXT DEFAULT 'active',
  full_investigation_log TEXT,
  assigned_staff TEXT,
  agency_fee NUMERIC DEFAULT 0,
  bonus_amount NUMERIC DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. TELECOM INTELLIGENCE
CREATE TABLE IF NOT EXISTS telecom_intelligence (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  case_id UUID REFERENCES investigation_cases(id),
  contact_name TEXT,
  phone_number TEXT,
  relation_type TEXT,
  associated_emails TEXT[],
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()

  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  role TEXT NOT NULL CHECK (role IN ('user','assistant','system')),
  content TEXT NOT NULL,
  agent TEXT DEFAULT 'sherlock',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. KNOWLEDGE GRAPH — NODES
CREATE TABLE IF NOT EXISTS kg_nodes (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  ref TEXT NOT NULL,
  entity_type TEXT,
  entity_name TEXT,
  attributes JSONB,
  domain TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 5. KNOWLEDGE GRAPH — EDGES

  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  source_id UUID REFERENCES kg_nodes(id) ON DELETE CASCADE,
  target_id UUID REFERENCES kg_nodes(id) ON DELETE CASCADE,
  relation_type TEXT,
  weight FLOAT DEFAULT 1.0,
  evidence TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
